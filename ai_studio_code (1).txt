import React, { useState, useRef, useEffect } from 'react';
import { Property, TourRoom } from '../types';
import {
  X,
  Compass,
  Maximize,
  Minimize,
  RotateCw,
  Sparkles,
  PhoneCall,
  BedDouble,
  Home,
  CheckCircle2,
  ChevronLeft,
  ChevronRight,
  Eye
} from 'lucide-react';

interface VirtualTour3DProps {
  property: Property;
  onClose: () => void;
  onBookVisit: (property: Property) => void;
}

export const VirtualTour3D: React.FC<VirtualTour3DProps> = ({
  property,
  onClose,
  onBookVisit,
}) => {
  const tourData = property.virtualTour3D;
  const rooms = tourData?.rooms || [];
  const [activeRoomIndex, setActiveRoomIndex] = useState(0);
  const [rotation, setRotation] = useState({ yaw: 0, pitch: 0 });
  const [isDragging, setIsDragging] = useState(false);
  const [dragStart, setDragStart] = useState({ x: 0, y: 0 });
  const [isFullscreen, setIsFullscreen] = useState(false);
  const [autoRotate, setAutoRotate] = useState(true);

  const currentRoom: TourRoom | undefined = rooms[activeRoomIndex];

  // Auto rotation loop
  useEffect(() => {
    if (!autoRotate || isDragging) return;
    const interval = setInterval(() => {
      setRotation((prev) => ({
        ...prev,
        yaw: (prev.yaw + 0.2) % 360,
      }));
    }, 40);
    return () => clearInterval(interval);
  }, [autoRotate, isDragging]);

  const handleMouseDown = (e: React.MouseEvent) => {
    setIsDragging(true);
    setDragStart({ x: e.clientX, y: e.clientY });
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    if (!isDragging) return;
    const deltaX = e.clientX - dragStart.x;
    const deltaY = e.clientY - dragStart.y;

    setRotation((prev) => ({
      yaw: (prev.yaw - deltaX * 0.4 + 360) % 360,
      pitch: Math.max(-45, Math.min(45, prev.pitch + deltaY * 0.2)),
    }));

    setDragStart({ x: e.clientX, y: e.clientY });
  };

  const handleMouseUp = () => {
    setIsDragging(false);
  };

  return (
    <div className="fixed inset-0 z-50 bg-black/95 backdrop-blur-md flex flex-col select-none animate-in fade-in duration-200">
      
      {/* Top Overlay Bar */}
      <div className="p-4 sm:p-6 flex items-center justify-between z-20 bg-gradient-to-b from-black/80 to-transparent">
        <div className="flex items-center gap-3">
          <div className="w-10 h-10 rounded-2xl bg-amber-500 text-slate-950 flex items-center justify-center font-bold shadow-lg">
            <Sparkles className="w-5 h-5" />
          </div>
          <div>
            <div className="flex items-center gap-2">
              <h2 className="text-white font-bold text-base sm:text-lg">{property.title}</h2>
              <span className="px-2 py-0.5 rounded-full bg-amber-500/20 text-amber-400 text-xs font-bold border border-amber-500/30">
                معاينة 360° تفاعلية
              </span>
            </div>
            <p className="text-slate-400 text-xs">{property.district} - {property.neighborhood}</p>
          </div>
        </div>

        <div className="flex items-center gap-2">
          <button
            onClick={() => setAutoRotate(!autoRotate)}
            className={`p-2.5 rounded-xl border text-xs font-bold flex items-center gap-1.5 transition-all ${
              autoRotate
                ? 'bg-amber-500 text-slate-950 border-amber-400'
                : 'bg-white/10 text-white border-white/20 hover:bg-white/20'
            }`}
            title="دوران تلقائي"
          >
            <RotateCw className={`w-4 h-4 ${autoRotate ? 'animate-spin' : ''}`} />
            <span className="hidden sm:inline">دوران تلقائي</span>
          </button>

          <button
            onClick={onClose}
            className="p-2.5 rounded-xl bg-white/10 hover:bg-white/20 text-white border border-white/20 transition-all"
            title="إغلاق المعاينة"
          >
            <X className="w-5 h-5" />
          </button>
        </div>
      </div>

      {/* Main 360 Canvas Viewer */}
      <div
        onMouseDown={handleMouseDown}
        onMouseMove={handleMouseMove}
        onMouseUp={handleMouseUp}
        className="flex-1 relative overflow-hidden cursor-grab active:cursor-grabbing flex items-center justify-center"
      >
        {currentRoom ? (
          <div
            className="w-[120%] h-[120%] absolute transition-transform duration-75 ease-out scale-105"
            style={{
              backgroundImage: `url(${currentRoom.panoramaUrl})`,
              backgroundPosition: `${(rotation.yaw / 360) * 100}% ${50 + rotation.pitch * 0.8}%`,
              backgroundSize: 'cover',
              filter: 'brightness(0.95)',
            }}
          >
            <div className="absolute inset-0 bg-radial from-transparent via-transparent to-black/40 pointer-events-none" />
          </div>
        ) : (
          <div className="text-white text-center">
            <p>لا تتوفر صور 360° لهذا العقار حالياً.</p>
          </div>
        )}

        {/* Floating Room Info Card */}
        {currentRoom && (
          <div className="absolute top-6 left-6 z-20 bg-black/60 backdrop-blur-md p-4 rounded-2xl border border-white/15 text-white max-w-xs space-y-2">
            <div className="flex items-center gap-2">
              <span className="w-2.5 h-2.5 rounded-full bg-emerald-400 animate-pulse" />
              <h3 className="font-bold text-sm text-amber-300">{currentRoom.name}</h3>
            </div>
            {currentRoom.dimensions && (
              <p className="text-xs text-slate-300">الأبعاد: {currentRoom.dimensions}</p>
            )}
            {currentRoom.features && (
              <div className="flex flex-wrap gap-1 pt-1">
                {currentRoom.features.map((f, i) => (
                  <span key={i} className="text-[10px] px-2 py-0.5 rounded bg-white/10 text-slate-200">
                    {f}
                  </span>
                ))}
              </div>
            )}
          </div>
        )}

        {/* Drag Instruction Banner */}
        <div className="absolute bottom-28 sm:bottom-24 bg-black/50 backdrop-blur-md px-4 py-1.5 rounded-full text-white/80 text-xs border border-white/10 pointer-events-none flex items-center gap-2">
          <span>اسحب الماوس أو المس الشاشة للتدوير بحرية 360° في الغرفة</span>
        </div>
      </div>

      {/* Bottom Room Selector Strip */}
      <div className="p-4 sm:p-6 bg-gradient-to-t from-black/90 via-black/80 to-transparent z-20">
        <div className="max-w-4xl mx-auto flex items-center justify-between gap-4">
          <div className="flex items-center gap-2 overflow-x-auto pb-2 sm:pb-0 scrollbar-none">
            {rooms.map((room, idx) => (
              <button
                key={room.id}
                onClick={() => setActiveRoomIndex(idx)}
                className={`flex items-center gap-2.5 p-2 rounded-2xl border transition-all shrink-0 ${
                  activeRoomIndex === idx
                    ? 'bg-amber-500 text-slate-950 border-amber-400 shadow-lg font-bold'
                    : 'bg-white/10 text-white border-white/10 hover:bg-white/20'
                }`}
              >
                <img
                  src={room.previewUrl || room.panoramaUrl}
                  alt={room.name}
                  className="w-10 h-10 rounded-xl object-cover"
                />
                <div className="text-right pl-2">
                  <div className="text-xs">{room.name}</div>
                  <div className="text-[10px] opacity-80">{room.dimensions || '360°'}</div>
                </div>
              </button>
            ))}
          </div>

          <button
            onClick={() => onBookVisit(property)}
            className="px-5 py-3 rounded-2xl bg-gradient-to-r from-amber-500 to-amber-600 hover:from-amber-400 hover:to-amber-500 text-slate-950 font-black text-xs sm:text-sm flex items-center gap-2 shadow-xl shrink-0"
          >
            <PhoneCall className="w-4 h-4" />
            <span>طلب معاينة ميدانية مع الدلال</span>
          </button>
        </div>
      </div>

    </div>
  );
};
